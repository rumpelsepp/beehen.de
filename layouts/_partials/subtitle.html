{{- /* Meta line of a post: date, tags and, if asked for, who wrote it. Params (dict): page the post author true to name the author, "author" in the front matter (default: false) */ -}} {{- $showAuthor := .author | default false -}} {{- with .page -}} {{- $dateTag := printf `%s` (.Date.Format "2006-01-02") (.Date.Format "02.01.2006") -}} {{- $subtitle := printf ` %s` $dateTag -}} {{- if $showAuthor -}} {{- with .Params.author -}} {{- $subtitle = printf `%s %s` $subtitle (. | htmlEscape) -}} {{- end -}} {{- end -}} {{- with .Params.tags -}} {{- $links := slice -}} {{- range . -}} {{- $tagName := . -}} {{- with site.GetPage (printf "tags/%s" $tagName) -}} {{- $links = $links | append (printf `#%s` .RelPermalink $tagName) -}} {{- else -}} {{- $links = $links | append (printf "#%s" $tagName) -}} {{- end -}} {{- end -}} {{- $subtitle = printf `%s %s` $subtitle (delimit $links " ") -}} {{- end -}} {{- printf `%s` $subtitle | safeHTML -}} {{- end -}}
