{{- /* Local override of Hugo's embedded _funcs/get-page-images.html. "RelPermalink" .RelPermalink "Permalink" .Permalink ) }} {{- else }} {{- $imgs = $imgs | append (dict "RelPermalink" (relURL $img) "Permalink" (absURL $img) ) }} {{- end }} {{- else }} {{/* External image */}} {{- $imgs = $imgs | append (dict "RelPermalink" $img "Permalink" $img ) }} {{- end }} {{- end }} {{- return $imgs }}
