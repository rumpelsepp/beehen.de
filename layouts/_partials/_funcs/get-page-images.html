{{- /* Local override of Hugo's embedded _funcs/get-page-images.html. Identical to the built-in, except the auto-discovered image: - resolves the *same* front-matter `image` parameter the home page's post cards use (layouts/home.html), defaulting to "feature.jpg" — so OG/Twitter/schema show the same picture as the site itself does. - downscales it to the Open Graph reference size (1200x630) instead of serving it at full resolution. - falls back to the site favicon (same resource head.html uses for the ) when a page has no featured image at all. opengraph.html, twitter_cards.html and schema.html all consume this helper, so they pick up the resized URL automatically. Images set explicitly via the `images` front-matter parameter or the site `params.images` are left untouched. */ -}} {{- $imgs := slice }} {{- $imgParams := .Params.images }} {{- $resources := .Resources.ByType "image" -}} {{/* Find the page's featured image (or fall back to the favicon) if the images parameter is empty. */}} {{- if not $imgParams }} {{- $featured := $resources.GetMatch (.Param "image" | default "feature.jpg") -}} {{- $og := "" -}} {{- if and $featured (reflect.IsImageResourceProcessable $featured) -}} {{- $og = ($featured.Filter images.AutoOrient).Process "fill 1200x630 q80" -}} {{- else -}} {{- with resources.Get "favicon.png" -}} {{- $og = .Resize "1200x jpg q80" -}} {{- end -}} {{- end -}} {{- with $og }} {{- $imgs = $imgs | append (dict "Image" . "RelPermalink" .RelPermalink "Permalink" .Permalink ) }} {{- end }} {{- end }} {{/* Use the first one of site images as the fallback. */}} {{- if and (not $imgParams) (not $imgs) }} {{- with site.Params.images }} {{- $imgParams = first 1 . }} {{- end }} {{- end }} {{/* Parse page's images parameter. */}} {{- range $imgParams }} {{- $img := . }} {{- $url := urls.Parse $img }} {{- if eq $url.Scheme "" }} {{/* Internal image: check page resources first, then global resources. */}} {{- with or ($resources.GetMatch $img) (resources.GetMatch $img) -}} {{- $imgs = $imgs | append (dict "Image" . "RelPermalink" .RelPermalink "Permalink" .Permalink ) }} {{- else }} {{- $imgs = $imgs | append (dict "RelPermalink" (relURL $img) "Permalink" (absURL $img) ) }} {{- end }} {{- else }} {{/* External image */}} {{- $imgs = $imgs | append (dict "RelPermalink" $img "Permalink" $img ) }} {{- end }} {{- end }} {{- return $imgs }}
